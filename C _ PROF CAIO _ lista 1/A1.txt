// faça um programa que leia três números e calcule a média - A1
//______________________________________________________________

#include<stdio.h>

int main (void){

    float nI; 
    float nII; 
    float nIII; 
    float MEdia;

     printf("Escreva sua primeira nota: \n");
     scanf("%f", &nI);

     printf("Escreva sua segunda nota: \n");
     scanf("%f", &nII);

     printf("Escreva sua terceira nota: \n");
     scanf("%f", &nIII);

     MEdia = (nI + nII + nIII) / 3;

     printf("A media das 3 notas eh: %f\n", MEdia);

     return 0;
    
}
